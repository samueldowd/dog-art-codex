from pathlib import Path
import argparse
import json
import subprocess
import sys

sys.path.insert(0, str(Path(__file__).parent / 'python'))
import imageio_ffmpeg
from PIL import Image, ImageDraw, ImageOps

parser = argparse.ArgumentParser()
parser.add_argument('video')
parser.add_argument('output')
args = parser.parse_args()
video = Path(args.video)
out = Path(args.output)
out.mkdir(parents=True, exist_ok=True)
reader = imageio_ffmpeg.read_frames(str(video))
metadata = next(reader)
reader.close()
(out / 'metadata.json').write_text(json.dumps(metadata, indent=2), encoding='utf-8')
print(json.dumps({'video': video.name, 'metadata': metadata}), flush=True)
duration = float(metadata['duration'])
subprocess.run([
    imageio_ffmpeg.get_ffmpeg_exe(), '-hide_banner', '-loglevel', 'error',
    '-i', str(video), '-vf', f'fps={9 / duration}', '-frames:v', '9',
    '-y', str(out / 'frame-%02d.png')
], check=True)
frames = sorted(out.glob('frame-*.png'))
sheet = Image.new('RGB', (1080, 900), '#e9e9e9')
draw = ImageDraw.Draw(sheet)
for i, frame in enumerate(frames):
    with Image.open(frame) as image:
        thumb = ImageOps.contain(image.convert('RGB'), (350, 270))
    x = (i % 3) * 360 + (360 - thumb.width) // 2
    y = (i // 3) * 300 + 22
    sheet.paste(thumb, (x, y))
    draw.text(((i % 3) * 360 + 12, (i // 3) * 300 + 5), f'Frame {i + 1}', fill='black')
sheet.save(out / 'contact-sheet.jpg', quality=95)
print(str((out / 'contact-sheet.jpg').resolve()), flush=True)
